% DZG Core Package Loading
% Shared preamble fragment included by dzg-unified.sty and dzg-tikz.sty.
% Package loading only — tier macro inclusions are each .sty's own concern.

% The math, graphics and diagram packages
\input{preambles/dzg-math-preamble}


% Graphics are found through TEXINPUTS, which the pandoc justfile points at
% ~/.pandoc and ~/.pandoc/figures. A \graphicspath entry cannot be used for
% this: TeX does not expand "~", so such an entry never matches anything.

% Microtype: protrusion and expansion.
% Pandoc's default template may load microtype before project styles.
\RequirePackage{microtype}
% kerning and spacing are pdfTeX-only features; microtype errors on spacing under LuaTeX.
\ifpdftex
  \microtypesetup{activate=true, final=true, tracking=true, kerning=true, spacing=true}
\else
  \microtypesetup{activate=true, final=true, tracking=true}
\fi

% Hyperlinks
\RequirePackage[colorlinks=true, linkcolor=blue, citecolor=blue, urlcolor=blue]{hyperref}
\RequirePackage{bookmark}

% List spacing
\RequirePackage{enumitem}
\setlist[enumerate]{itemsep=2pt,parsep=2pt,before={\parskip=2pt}}
