% =============================================================================
% Baily-Borel cusp diagrams of moduli spaces
% =============================================================================
% Every cusp diagram is a pic. A 0-cusp (the orbit of a primitive isotropic
% line) is a cusp0 node and a 1-cusp (the orbit of a primitive isotropic plane)
% a cusp1 node; an incidence arrow runs from a 0-cusp to each 1-cusp whose
% closure contains it. Node names are the canonical names listed below, so
%   \pic (S) {bb fen2};
% names the 0-cusps S1, ..., S5 and the 1-cusps S12, S245, .... Decorate
% through the node centres, e.g. the image of another boundary:
%   \scoped[on background layer] \draw[cusp image] (S2.center) -- (S245.center);
%
% Keys:
%   cusp labels=canonical   the labels below (default)
%   cusp labels=eta         the names eta_i, I_{i,j} of the dissertation text
%                           (Enriques-K3 chapter) where the object has them
%   cusp labels=none        no labels
%   incidence divisibility  double each incidence whose second isotropic
%                           vector has divisibility 2 in the 0-cusp lattice
%                           (an even move); a single arrow is divisibility 1.
%                           For 2-elementary lattices this is the incidence at
%                           which the a-invariant drops by 2 [AE22, Prop. 5.5].
%
% Published names and data:
%   bb fen2    Sterk's F_{En,2}             [Ste91 §4]; 0-cusps 1..5,
%              1-cusps named by the 0-cusps in their closure
%   bb f220    F_{(2,2,0)}, quartic hyperelliptic K3   [AE22 §10]
%   bb fen     unpolarized Enriques F_En    [AEGS25], [Ste91 §4.2]
%   bb fco     unpolarized Coble F_Co       (writing/coble/cusp-correspondence)
%   bb f2, bb f4, bb fell     degree 2 and 4 K3, elliptic K3   [Sca87 §6]
%   bb f2d=1|3|5|p            Scattone's F_{2d} by the class of d   [Sca87 §6]
%
% Lattice names. A 0-cusp or 1-cusp is named by its boundary lattice
% I^perp/I. Where the lattice has a standard name it is used (E_{10}(2) =
% U(2)+E_8(2), D_8^*(2), D_{16}^+, A_1^{+7}); the (16,2,0) lattices of
% F_{(2,2,0)} have none and carry Sterk's names, the root system of their
% (-2)-roots and reflective (-4)-roots (C_2, A_1(2) contain (-4)-roots),
% which determines the lattice in its genus. Computed with Sage (Gram
% matrices of catalogue.py TEn, TdP, Tco; vectors of sterk.py
% selected_isotropic_vectors and coble.py; 2026-09-24):
%   Sterk cusp      1          2          3, 4, 5
%   div in T_En     1          2          2
%   eta^perp/eta    (10,10,0)  (10,8,0)   (10,8,0)
%   div in T_dP     1          2          1
%   eta^perp/eta    (18,2,0)   (18,0,0)   (18,2,0)
% so cusp 2 lies over U+E_8^2 and cusps 1, 3, 4, 5 over U(2)+E_8^2. In T_En,
% <e,e'>^perp/<e,e'> = E_8(2), with 240 (-4)-roots, and
% <e',2e+2f+a1+a2>^perp/.. = (8,6,0), whose roots form B_8 (16 short, 112
% long): the index-2 even overlattice of A_1^8, i.e. D_8^*(2). The Coble plane
% <e', 2h+a1+a2> has divisibilities (2,2) in T_Co and T_En and (2,1) in T_dP,
% quotients (9,9,1), (7,7,1) in T_Co and (10,8,0), (8,6,0) in T_En; in T_dP its
% quotient is (16,0,0) with one irreducible root system of 480 roots, D_{16}^+.

\def\dzg@bb@scheme{canonical}
\tikzset{
  cusp labels/.is choice,
  cusp labels/canonical/.code={\def\dzg@bb@scheme{canonical}},
  cusp labels/eta/.code={\def\dzg@bb@scheme{eta}},
  cusp labels/none/.code={\def\dzg@bb@scheme{none}},
  incidence divisibility/.style={even incidence/.style={doubled mark}},
  even incidence/.style={},
}

% \dzg@bb@pick{<canonical>}{<eta>}: the text of the active scheme.
\def\dzg@bb@canonical{canonical}
\def\dzg@bb@eta{eta}
\def\dzg@bb@none{none}
\def\dzg@bb@pick#1#2{\csname dzg@bb@pick@\dzg@bb@scheme\endcsname{#1}{#2}}
\def\dzg@bb@pick@canonical#1#2{#1}
\def\dzg@bb@pick@eta#1#2{#2}
\def\dzg@bb@pick@none#1#2{}

% \dzg@bb@cusp{<cusp0|cusp1>}{<name>}{<coordinate>}{<content>}
%   {<label direction>}{<canonical label>}{<eta label>}
% Content and labels are math; an empty label in the active scheme draws nothing.
\def\dzg@bb@cusp#1#2#3#4#5#6#7{%
  \ifx\dzg@bb@scheme\dzg@bb@canonical\def\dzg@bb@text{#6}\fi
  \ifx\dzg@bb@scheme\dzg@bb@eta\def\dzg@bb@text{#7}\fi
  \ifx\dzg@bb@scheme\dzg@bb@none\let\dzg@bb@text\@empty\fi
  \ifx\dzg@bb@text\@empty
    \node[#1] (#2) at (#3) {$#4$};%
  \else
    \node[#1, label={[cusp label]#5:$\dzg@bb@text$}] (#2) at (#3) {$#4$};%
  \fi}

% \dzg@bb@incidences{<0-cusp>/<1-cusp>/<divisibility of the second vector>, ...}
\def\dzg@bb@incidences#1{%
  \foreach \dzg@p/\dzg@c/\dzg@d in {#1}{%
    \ifnum\dzg@d=2
      \draw[incidence, even incidence] (\dzg@p) -- (\dzg@c);%
    \else
      \draw[incidence] (\dzg@p) -- (\dzg@c);%
    \fi}}

% \dzg@bb@brace{<x from>}{<x to>}{<y>}{<text>}: invariants under a column,
% shown in the canonical scheme.
\def\dzg@bb@brace#1#2#3#4{%
  \ifx\dzg@bb@scheme\dzg@bb@canonical
    \draw[cusp brace] (#1,#3) -- (#2,#3) node[midway, below=5.5ex] {$#4$};%
  \fi}

\def\dzg@bb@pic#1{\tikzset{pics/#1/.style={/tikz/transform shape, code={%
  \tikzset{transform shape=false}\csname dzg@bb@#1\endcsname}}}}

% bb f2d=<N>: the shape of Scattone's diagram of F_{2d} for the class N of d,
% N = 1, 3, 5, or p (an odd prime) [Sca87 §6]: a 0-cusp p_0 meeting the
% 1-cusps C_i over H/SL_2(Z), and further 0-cusps p_j meeting the 1-cusps
% C_i^{(N)} over H/Gamma'(N).
% -----------------------------------------------------------------------------

  \tikzset{transform shape=false}\csname dzg@bb@f2d@#1\endcsname}}}
% \dzg@bb@column{<x>}{<node prefix>}{<first>}{<second>}{<next to last>}{<last>}
\def\dzg@bb@column#1#2#3#4#5#6{%
  \foreach \dzg@k/\dzg@y/\dzg@L in {1/2/{#3}, 2/1/{#4}, 3/-1/{#5}, 4/-2/{#6}}
    {\dzg@bb@cusp{cusp1}{#2\dzg@k}{#1,\dzg@y}{\dzg@L}{}{}{}}%
  \node at (#1,0) {$\vdots$};}
\@namedef{dzg@bb@f2d@1}{%
  \dzg@bb@column{0}{C}{C_1}{C_2}{C_{n-1}}{C_n}%
  \dzg@bb@cusp{cusp0}{P0}{3,0}{p_0}{}{}{}%
  \node at (0,-3) {$\mathbb{H}/\operatorname{SL}_2(\mathbb{Z})$};
  \foreach \dzg@k in {1,...,4} {\dzg@bb@incidences{P0/C\dzg@k/1}}}
\@namedef{dzg@bb@f2d@3}{%
  \dzg@bb@column{0}{C}{C_1}{C_2}{C_{n-1}}{C_n}%

  \dzg@bb@cusp{cusp0}{P0}{2.5,0}{p_0}{}{}{}%
  \dzg@bb@cusp{cusp0}{P1}{8,0}{p_1}{}{}{}%

  \node at (5,-3) {$\mathbb{H}/\Gamma'(3)$};
  \foreach \dzg@k in {1,...,4}
    {\dzg@bb@incidences{P0/C\dzg@k/1, P0/D\dzg@k/1, P1/D\dzg@k/1}}}
\@namedef{dzg@bb@f2d@5}{%
  \dzg@bb@column{0}{C}{C_1}{C_2}{C_{n-1}}{C_n}%
