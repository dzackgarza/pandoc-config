% =============================================================================
% Hasse diagrams of serialized posets: constructors
% =============================================================================
% \posetfromjson[<keys>]{<file.json>}: the Hasse diagram of the poset in a
% networkx node-link JSON file, which Sage writes for any finite poset with
% bin/poset_json.sage. The layout (lua/dzg-poset.lua) grades the poset, lets
% Graphviz dot order and space the elements of each grade, and places every
% element at the height of its grade. Each element is a node named by its id
% (so ids must be TikZ node names), an `element width` x `element height`
% box holding its label; \posetelements lists the ids, so a figure can place
% arbitrary content (a pic, a scope) in every box after the layout:
%   \posetfromjson[element width=2cm, element height=1.5cm]{m2-strata.json}
%   \foreach \e in \posetelements \pic at (\e) {object=moduli/m2-graph-\e};
% Keys:
%   poset grading=<method>   used when the elements carry no `grade`:
%                            balanced (default), longest chain from bottom,
%                            longest chain to top, minimum total cover length
%                            (lua/dzg-poset.lua defines and compares them)
%   poset axis=vertical|horizontal
%                            grades bottom to top (default), or right to left
%                            with the top element leftmost
%   grade distance=<length>  the distance per unit of grade
%   element width, element height   the box every element's content fits in
%   element sep=<length>     the least gap between two boxes of one grade
%   element positions=dot|even
%                            dot's coordinates (default: covers as vertical as
%                            dot can make them, rows with gaps; best for
%                            sparse or irregular posets), or dot's
%                            order within each grade, evenly spaced and
%                            centred (compact rows; best when every grade is
%                            a full row, as in Boolean and face lattices, and
%                            too steep for sparse posets such as Tamari)
%   element style=<style>    added to each element node (default none)
%   element labels=true|false   the JSON labels in the boxes (default true)
%   element layer=<layer>    the layer of the element nodes (default main;
%                            background puts filled cells under content
%                            that draws on the background layer)
%   cover layer=<layer>      the layer of the covers (default edges)
%   cover style=<style>      each cover, drawn from the larger element to
%                            the smaller (default poset cover; degeneration
%                            for strata posets, AGENTS.md rule 3)
%   cover direction=down|up  covers drawn from the larger element to the
%                            smaller (default), or from the smaller to the
%                            larger
% Other keys apply to the scope holding the diagram.
% LuaLaTeX with --shell-escape only: the layout runs Graphviz dot.

\def\poset@grading{balanced}
\def\poset@axis{vertical}
\def\poset@distance{1cm}
\def\posetelementwidth{1cm}
\def\posetelementheight{0.6cm}
\def\poset@sep{3mm}
\def\poset@element{}
\def\poset@cover{poset cover}
\def\poset@direction{down}
\def\poset@layer{main}
\def\poset@coverlayer{edges}
\def\poset@positions{dot}
\newif\ifposet@labels \poset@labelstrue
\tikzset{
  poset grading/.store in=\poset@grading,
  poset axis/.store in=\poset@axis,
  grade distance/.store in=\poset@distance,
  element width/.store in=\posetelementwidth,
  element height/.store in=\posetelementheight,
  element sep/.store in=\poset@sep,
  element style/.store in=\poset@element,
  cover style/.store in=\poset@cover,
  cover direction/.store in=\poset@direction,
  element layer/.store in=\poset@layer,
  cover layer/.store in=\poset@coverlayer,
  element positions/.store in=\poset@positions,
  element labels/.is if=poset@labels,
  poset element/.style={shape=rectangle, inner sep=0pt,
    minimum width=\posetelementwidth, minimum height=\posetelementheight},
  poset cover/.style={draw, line width=0.6pt, -},
}
\ifdefined\directlua
  \directlua{dzgposet = require("dzg-poset")}
  \newcommand\posetfromjson[2][]{%
    \begingroup
    % Read the settings; the transformations among #1 apply once, in the
    % scope the layout emits.
    \pgfgettransform\poset@transform
    \tikzset{#1}%
    \pgfsettransform\poset@transform
    \pgfmathsetmacro\poset@w{(\posetelementwidth)/1cm}%
    \pgfmathsetmacro\poset@h{(\posetelementheight)/1cm}%
    \pgfmathsetmacro\poset@s{(\poset@sep)/1cm}%
    \pgfmathsetmacro\poset@d{(\poset@distance)/1cm}%
    \directlua{dzgposet.emit{path = "\luaescapestring{#2}",
      options = "\luaescapestring{\unexpanded{#1}}",
      grading = "\luaescapestring{\poset@grading}",
      axis = "\luaescapestring{\poset@axis}",
      element = "\luaescapestring{\unexpanded\expandafter{\poset@element}}",
      cover = "\luaescapestring{\unexpanded\expandafter{\poset@cover}}",
      labels = \ifposet@labels true\else false\fi,
      direction = "\luaescapestring{\poset@direction}",
      layer = "\luaescapestring{\poset@layer}",
      cover_layer = "\luaescapestring{\poset@coverlayer}",
      positions = "\luaescapestring{\poset@positions}",
      width = \poset@w, height = \poset@h, sep = \poset@s, distance = \poset@d}}%
    \endgroup}
\else
  \newcommand\posetfromjson[2][]{\PackageError{dzg}{\string\posetfromjson\space
    needs LuaLaTeX}{The layout runs Graphviz dot from Lua; render the figure
    with lualatex --shell-escape.}}
\fi
