% =============================================================================
% TikZ library "dzg": the figure vocabulary of this repository.
% =============================================================================
% Loaded by \usetikzlibrary{dzg} in styles/preambles/dzg-preamble.tex, so every
% document, every standalone figure, and every figure the tikzcd filter compiles
% sees the same styles. Figures use the semantic style names below and never
% redefine them; a figure that needs a new element adds it here.
%
% Every size is an absolute length. Node sizes therefore do not change with a
% picture's `scale`, which only spaces the coordinates.
%
% Families:
%   colours ........ dzg accent, dzg singular, dzg highlight, dzg region, ...
%   Coxeter ........ vertex=white|black|doubled, coxeter edge=3|4|5|6|infinity|
%                    dotted, coxeter edge label, parabolic, vertex label,
%                    inactive, marked vertex, subdiagram cell=elliptic|
%                    parabolic|other, fold, fold axis
%   Baily-Borel .... cusp0, cusp1, incidence, cusp brace
%   IAS / Kulikov .. ias region, ias boundary, ias singularity, ias surgery,
%                    fan ray, kulikov component, double curve
%   lattice ........ lattice grid, lattice point, polygon region, long side,
%                    short side, axis side, boundary point, marked point,
%                    side label, monomial, polygon title, polytope edge,
%                    distinguished point, coordinate axis, hatched region
%   posets, moduli . poset node, degeneration, morphism, poset group,
%                    moduli blob, stratum, boundary stratum,
%                    mirror move=complement|U|U(2)
%   curves ......... curve component, curve node, dual vertex, genus,
%                    curve point, singular point, constructed point
%   spaces, maps ... point, map arrow, distinguished curve, contour, over strand,
%                    branch cut, glued edge=1|2, complex plane (pgfplots axis)
% =============================================================================

\usetikzlibrary{arrows.meta, backgrounds, calc, decorations.pathreplacing,
  decorations.markings, decorations.pathmorphing, fit, hobby, positioning, shapes.geometric, shapes.misc}

% -----------------------------------------------------------------------------
% Layers
% -----------------------------------------------------------------------------
% dynkin-diagrams declares background and "Dynkin behind"; the backgrounds
% library declares background. Diagram edges go on the edges layer so vertices
% cover their ends, and highlights go on background under both. Never call
% \pgfsetlayers in a figure.
\pgfutil@ifundefined{pgf@layerbox@background}{\pgfdeclarelayer{background}}{}
\pgfutil@ifundefined{pgf@layerbox@Dynkin behind}{\pgfdeclarelayer{Dynkin behind}}{}
\pgfdeclarelayer{edges}
\pgfdeclarelayer{foreground}
\pgfsetlayers{background,Dynkin behind,edges,main,foreground}

% Scope keys for the two extra layers, after `on background layer` in
% tikzlibrarybackgrounds.code.tex:
%   \begin{scope}[on edge layer] \draw[coxeter edge=4] (a) -- (b); \end{scope}
\tikzset{
  on edge layer/.style={
    execute at begin scope={\pgfonlayer{edges}\let\tikz@options=\pgfutil@empty
      \tikzset{#1}\tikz@options},
    execute at end scope={\endpgfonlayer}},
  on foreground layer/.style={
    execute at begin scope={\pgfonlayer{foreground}\let\tikz@options=\pgfutil@empty
      \tikzset{#1}\tikz@options},
    execute at end scope={\endpgfonlayer}},
}

% -----------------------------------------------------------------------------
% Colours
% -----------------------------------------------------------------------------
\colorlet{dzg accent}{blue!70!black}      % rays, maps, distinguished curves
\colorlet{dzg singular}{red!75!black}     % singular points, surgeries
\colorlet{dzg highlight}{cyan!30}         % parabolic subdiagrams, emphasis
\colorlet{dzg region}{cyan!8}             % filled regions of polygons and IAS
\colorlet{dzg grid}{gray!40}              % coordinate and lattice grids
\colorlet{dzg muted}{black!45}            % secondary strokes and points
\colorlet{dzg blob}{black!8}              % moduli-space schematics
\colorlet{dzg elliptic}{green!14}         % elliptic subdiagrams (finite type)
\colorlet{dzg parabolic}{yellow!24}       % maximal parabolic subdiagrams

% -----------------------------------------------------------------------------
% Global defaults
% -----------------------------------------------------------------------------
\tikzset{
  >={Latex[length=2mm]},
  % Round caps and joins for strokes that meet at corners (polygon sides,
  % curves), applied per style so tikz-cd double arrows keep butt caps.
  round stroke/.style={line cap=round, line join=round},
}

% -----------------------------------------------------------------------------
% Coxeter, Dynkin and Vinberg diagrams
% -----------------------------------------------------------------------------
% The vertex mark is the datum. Its meaning is set by the document: in the
% Coble and Enriques writing white is a root of square -2 and black a root of
% square -4; in the Nikulin tables the marks encode delta.
\tikzset{
  vertex/.style={vertex/#1},
  vertex/base/.style={circle, draw, line width=0.6pt, inner sep=0pt,
    minimum size=5.5pt},
  vertex/white/.style={vertex/base, fill=white},
  vertex/black/.style={vertex/base, fill=black},
  vertex/doubled/.style={vertex/base, fill=white, line width=0.5pt,
    double, double distance=1.2pt, minimum size=6.5pt},
  vertex label/.style={font=\scriptsize, inner sep=1.5pt},
  %
  % Coxeter weight m of the edge; m=3 is the simple edge.
  coxeter edge/.style={coxeter edge/#1},
  coxeter edge/3/.style={draw, line width=0.6pt},
  coxeter edge/4/.style={draw, line width=0.5pt, double, double distance=1.6pt},
  coxeter edge/5/.style={coxeter edge/4, postaction={draw, line width=0.5pt}},
  coxeter edge/6/.style={draw, line width=0.5pt, double, double distance=3.4pt,
    postaction={draw, line width=0.5pt, double, double distance=0.6pt}},
  coxeter edge/infinity/.style={draw, line width=2pt},
  coxeter edge/dotted/.style={draw, line width=0.8pt, dash pattern=on 2pt off 2pt},
  % m >= 7: a simple edge carrying its weight.
  coxeter edge label/.style={coxeter edge/3, postaction={decorate,
    decoration={markings, mark=at position 0.5 with
      {\node[above, font=\scriptsize, inner sep=1.5pt] {$#1$};}}}},
  %
  % A parabolic subdiagram is highlighted by stroking its edges, and a zero-
  % length segment at each isolated vertex, under the diagram. Use the node
  % centres so the stroke is one unbroken path:
  %   \scoped[on background layer]
  %     \draw[parabolic] (a.center)--(b.center)--(c.center) (d.center)--(d.center);
  parabolic/.style={draw=dzg highlight, line width=12pt, line cap=round,
    line join=round},
  % A fold of a diagram onto its quotient.
  fold/.style={dzg singular, dashed, line width=1pt, <->},
  % Vertices and edges of an ambient diagram that lie outside the subdiagram
  % being shown. Apply to a scope, or before vertex= on a single vertex:
  %   \node[inactive, vertex=black] ...;  \draw[inactive, coxeter edge=4] ...;
  inactive/.style={draw=dzg grid, text=dzg grid,
    vertex/black/.append style={fill=dzg grid}},
  % A distinguished root (e.g. the root delta whose complement is taken),
  % drawn on the background layer behind its vertex.
  marked vertex/.style={circle, fill=dzg accent, fill opacity=0.5,
    inner sep=0pt, minimum size=10pt},
  % A cell of a subdiagram poset, coloured by the type of its subdiagram:
  % elliptic (finite), maximal parabolic (affine of maximal rank), or other.
  subdiagram cell/.style={subdiagram cell/#1},
  subdiagram cell/base/.style={draw, rounded corners=2pt, line width=0.5pt},
  subdiagram cell/elliptic/.style={subdiagram cell/base, fill=dzg elliptic},
  subdiagram cell/parabolic/.style={subdiagram cell/base, fill=dzg parabolic},
  subdiagram cell/other/.style={subdiagram cell/base, fill=white},
  % The fixed line of a reflection symmetry of a diagram.
  fold axis/.style={draw=dzg singular, dotted, line width=0.8pt},
}

% -----------------------------------------------------------------------------
% Baily-Borel cusp diagrams
% -----------------------------------------------------------------------------
% 0-cusp: rounded box; 1-cusp: box. The lattice goes in the node, invariants in
% `label`. An incidence arrow joins a 1-cusp to a 0-cusp in its closure.
\tikzset{
  cusp0/.style={rounded rectangle, draw, line width=0.6pt, minimum width=1.6cm,
    minimum height=6mm, inner xsep=3pt},
  cusp1/.style={rectangle, draw, line width=0.6pt, minimum width=1.4cm,
    minimum height=6mm, inner xsep=3pt},
  cusp label/.style={font=\scriptsize, inner sep=1pt, label distance=0.5em},
  incidence/.style={->, line width=0.6pt, shorten >=1pt, shorten <=1pt},
  cusp brace/.style={decorate, decoration={brace, amplitude=5pt, mirror,
    raise=4ex}},
}

% -----------------------------------------------------------------------------
% Integral-affine spheres, fans, Kulikov models
% -----------------------------------------------------------------------------
\tikzset{
  ias region/.style={fill=dzg region},
  ias boundary/.style={draw, line width=0.8pt, round stroke},
  ias edge/.style={draw, line width=0.5pt},
  % Singular point; the optional value is its multiplicity, set as a label:
  %   \node[ias singularity=16] at (p) {};
  ias singularity/.style={star, star points=5, star point ratio=2.4,
    draw=dzg singular, fill=white, line width=0.8pt, inner sep=0pt,
    minimum size=9pt,
    label={[font=\footnotesize, text=dzg singular, inner sep=1pt]above right:#1}},
  ias singularity/.default={},
  ias surgery/.style={fill=dzg singular!35, draw=dzg singular, line width=0.6pt},
  fan ray/.style={->, draw=dzg accent, line width=1pt},
  kulikov component/.style={draw, line width=0.6pt, fill=dzg region},
  double curve/.style={draw, line width=1.4pt, round stroke},
  nontoric blowup/.style={circle, fill=dzg singular, inner sep=0pt,
    minimum size=4pt},
}

% -----------------------------------------------------------------------------
% Lattice polygons (toric and ADE surfaces)
% -----------------------------------------------------------------------------
% Sides of an ADE polygon: `long side` and `short side` are sides of the
% boundary divisor of lattice length >= 2 and 1; `axis side` is a side on a
% coordinate axis, off the boundary divisor, along which the ADE chain lies.
% `side label` sets a brace and its text outside a side traversed with the
% polygon on its left (counterclockwise):
%   \path[side label=2] (4,0) -- (0,2);
\tikzset{
  lattice grid/.style={draw=dzg grid, line width=0.3pt},
  lattice point/.style={circle, fill=dzg muted, inner sep=0pt, minimum size=3pt},
  polygon region/.style={fill=dzg region},
  long side/.style={draw, line width=2pt, round stroke},
  short side/.style={draw, line width=1pt, round stroke},
  axis side/.style={draw=dzg muted, line width=0.8pt},
  % A lattice point on the boundary divisor; the value is its fill.
  boundary point/.style={circle, draw, fill=#1, line width=0.8pt,
    inner sep=0pt, minimum size=6pt},
  boundary point/.default=white,
  marked point/.style={ias singularity=#1},
  marked point/.default={},
  side label/.style={
    postaction={draw=dzg accent, line width=0.5pt, decorate,
      decoration={brace, amplitude=4pt, mirror, raise=6mm}},
    postaction={decorate, decoration={markings, mark=at position 0.5 with
      {\node[text=dzg accent, font=\small, inner sep=1pt] at (0,-10mm) {#1};}}}},
  % The monomial x^i y^j written at the lattice point (i,j).
  monomial/.style={font=\scriptsize, inner sep=4.5pt},
  polygon title/.style={font=\large, inner sep=2pt},
  % An edge of a lattice polytope (moment polytope, polytope of a fan).
  polytope edge/.style={draw, line width=0.8pt, round stroke},
  % A lattice point the text singles out: a polytope vertex, a cone generator,
  % the origin of a fan; `lattice point` is the ambient lattice around it.
  distinguished point/.style={circle, fill=black, inner sep=0pt,
    minimum size=4.5pt},
  % A coordinate axis of the ambient real vector space or affine plane.
  coordinate axis/.style={->, draw=dzg muted, line width=0.4pt},
  % A cone or region marked by hatching, where filled regions would overlap.
  hatched region/.style={pattern=north east lines, pattern color=dzg muted},
}

% -----------------------------------------------------------------------------
% Posets and moduli-space schematics
% -----------------------------------------------------------------------------
% Arrows point from the generic stratum to the more degenerate one.
\tikzset{
  poset node/.style={draw, rounded corners=2pt, line width=0.5pt,
    inner xsep=4pt, inner ysep=3pt, font=\small},
  degeneration/.style={->, line width=0.6pt, shorten >=3pt, shorten <=3pt},
  moduli blob/.style={fill=dzg blob, draw, line width=0.8pt, round stroke},
  stratum/.style={circle, fill=black, inner sep=0pt, minimum size=4pt},
  % Mirror moves of 2-elementary lattices (Alexeev-Engel): a lattice S to its
  % orthogonal complement T, then moves T -> e^perp/e of type U or U(2).
  mirror move/.style={mirror move/#1},
  mirror move/complement/.style={->, decorate, decoration={snake,
    amplitude=0.6mm, segment length=3mm, post length=2mm}},
  mirror move/U/.style={degeneration},
  mirror move/U(2)/.style={degeneration, double, double distance=1.2pt},
  boundary stratum/.style={draw=dzg singular, line width=1.4pt},
  % A morphism of spaces, e.g. the contraction of one compactification onto
  % another.
  morphism/.style={->, line width=0.9pt, shorten >=2pt, shorten <=2pt},
  % A box around a sub-poset: \node[poset group, fit=(a)(b)(c)] {};
  % draw it on the background layer.
  poset group/.style={draw, rounded corners=4pt, line width=0.5pt,
    fill=dzg region, inner sep=8pt},
}

% -----------------------------------------------------------------------------
% Stable curves and dual graphs
% -----------------------------------------------------------------------------
% Draw components with Hobby curves: to[curve through={...}] or closed hobby.
\tikzset{
  curve component/.style={draw, line width=0.9pt, round stroke},
  curve node/.style={circle, fill=black, inner sep=0pt, minimum size=3.5pt},
  dual vertex/.style={circle, fill=black, inner sep=0pt, minimum size=5pt},
  genus/.style={font=\scriptsize, inner sep=1pt},
  % A given point on a curve.
  curve point/.style={circle, fill=black, inner sep=0pt, minimum size=4pt},
  % A singular point of a plane curve (cusp, node, tacnode, ...).
  singular point/.style={circle, draw, fill=white, line width=0.8pt,
    inner sep=0pt, minimum size=6pt},
  % A point constructed from the given ones, e.g. an intersection of lines.
  constructed point/.style={circle, draw, fill=white, line width=0.8pt,
    inner sep=0pt, minimum size=4pt},
}

% -----------------------------------------------------------------------------
% Spaces, maps and contours (topology and complex-analysis pictures)
% -----------------------------------------------------------------------------
\tikzset{
  % A point of a space: a basepoint, an endpoint, a pole, a point of a fibre.
  point/.style={circle, fill, inner sep=0pt, minimum size=4pt},
  % A continuous map between two drawn spaces.
  map arrow/.style={->, line width=0.8pt},
  % A curve singled out in a space: a loop, a path, a fibre, a factor circle.
  distinguished curve/.style={draw=dzg accent, line width=1pt, round stroke},
  % An oriented path or integration contour, with arrow tips at the listed fractions
  % of its length: \draw[contour] ...; \draw[contour={0.2,0.7}] ...;
  contour/.style={draw, line width=0.8pt, postaction={decorate,
    decoration={markings, reset marks}, contour mark/.list={#1}}},
  contour/.default={0.5},
  contour mark/.style={decoration={mark=at position #1 with {\arrow{>}}}},
  % A strand passing over the strands drawn before it: a white halo cuts the
  % strand beneath at each crossing.
  over strand/.style={preaction={draw=white, line width=4pt, line cap=butt}},
  % A branch cut of a multivalued function.
  branch cut/.style={draw=dzg singular, line width=1.4pt},
  % An edge of a fundamental polygon; the number of arrow tips (1 or 2) names
  % the class of edges it is glued to.
  glued edge/.style={glued edge/#1},
  glued edge/1/.style={draw, line width=1.4pt, -{Stealth[length=3mm]}},
  glued edge/2/.style={draw, line width=1.4pt,
    -{Stealth[length=3mm] Stealth[length=3mm]}},
}
