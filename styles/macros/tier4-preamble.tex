\makeatletter
\def\@tocline#1#2#3#4#5#6#7{\relax
  \ifnum #1>\c@tocdepth \else
    \par \addpenalty\@secpenalty\addvspace{#2}\begingroup \hyphenpenalty\@M
    \@ifempty{#4}{\@tempdima\csname r@tocindent\number#1\endcsname\relax
    }{\@tempdima#4\relax
    }\parindent\z@ \leftskip#3\relax \advance\leftskip\@tempdima\relax
    \rightskip\@pnumwidth plus4em \parfillskip-\@pnumwidth
    #5\leavevmode\hskip-\@tempdima
      \ifcase #1
       \or\or \hskip 2em \or \hskip 3em \else \hskip 4em \fi#6\nobreak\relax
    \hfill\hbox to\@pnumwidth{\@tocpagenum{#7}}\par\nobreak
    \endgroup
  \fi}
\makeatother
