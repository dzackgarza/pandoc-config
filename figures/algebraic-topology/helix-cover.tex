\documentclass[border=10pt]{standalone}
\usepackage{dzg-tikz}
\begin{document}
\begin{tikzpicture}
  \def\R{1.9}      % helix radius (horizontal)
  \def\vs{0.34}    % vertical rise per radian
  \def\e{0.42}     % perspective squash
  % the helix R
  \draw[distinguished curve, smooth, samples=240, domain=0:25.133]
    plot ({\R*cos(deg(\x))},{\vs*\x+\e*sin(deg(\x))});
  \node[dzg accent] at (-2.55,7.7) {$\mathbb{R}$};
  % the fibre over the basepoint: the points 2 pi k, where cos = 1
  \draw[dzg singular, dashed] (\R,-1.55) -- (\R,{\vs*(2*pi*4)+0.3});
  \foreach \k in {0,...,4}
    \node[point, dzg singular] at (\R,{\vs*(2*pi*\k)}) {};
  \node[dzg singular, right] at (\R+0.25,{\vs*(2*pi*4)}) {$\mathbb{Z}=p^{-1}(x_0)$};
  % the circle S^1
  \begin{scope}[shift={(0,-2.0)}]
    \draw[curve component] (0,0) ellipse [x radius=\R, y radius=\e];
    \node[point, label={below right:$x_0$}] at (\R,0) {};
    \node at (-2.55,0.05) {$S^1$};
  \end{scope}
  % the covering map
  \draw[map arrow] (\R+0.95,2.4) -- (\R+0.95,0) node[midway, right] {$p$};
\end{tikzpicture}
\end{document}
