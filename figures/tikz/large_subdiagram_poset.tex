% Rounded convex capsule/hull path around a cyclic list of coordinates.
% For two points this gives a capsule around the segment.
% For one point use \GlowVertex.
\makeatletter
\def\convexpath#1#2{%
  [create hullcoords/.code={%
    \global\edef\namelist{#1}%
    \foreach [count=\counter] \nodename in \namelist {%
      \global\edef\numberofnodes{\counter}%
      \coordinate (hullcoord\counter) at (\nodename);%
    }%
    \coordinate (hullcoord0) at (hullcoord\numberofnodes);%
    \pgfmathtruncatemacro\lastnumber{\numberofnodes+1}%
    \coordinate (hullcoord\lastnumber) at (hullcoord1);%
  }, create hullcoords]
  ($(hullcoord1)!#2!-90:(hullcoord0)$)
  \foreach [
    evaluate=\currentnode as \previousnode using \currentnode-1,
    evaluate=\currentnode as \nextnode using \currentnode+1
  ] \currentnode in {1,...,\numberofnodes} {%
    let \p1 = ($(hullcoord\currentnode) - (hullcoord\previousnode)$),
        \n1 = {atan2(\y1,\x1) + 90},
        \p2 = ($(hullcoord\nextnode) - (hullcoord\currentnode)$),
        \n2 = {atan2(\y2,\x2) + 90},
        \n{delta} = {-Mod(\n1-\n2,360)}
    in {arc [start angle=\n1, delta angle=\n{delta}, radius=#2]}
    -- ($(hullcoord\nextnode)!#2!-90:(hullcoord\currentnode)$)%
  }%
}
\makeatother

% Main-layer glow.
% It is drawn after the faded ambient diagram and before the final active
% black strokes/nodes.
\newcommand{\GlowSegment}[2]{%
  \fill[glowblue, opacity=.72]
    \convexpath{#1,#2}{5.8pt} -- cycle;
  \draw[cyan!55, opacity=.48, line width=1.05pt]
    \convexpath{#1,#2}{5.8pt} -- cycle;
}

\newcommand{\GlowVertex}[1]{%
  \fill[glowblue, opacity=.72]
    (#1) circle[radius=6.1pt];
  \draw[cyan!55, opacity=.48, line width=1.05pt]
    (#1) circle[radius=6.1pt];
}

% \SubdiagramCell{name}{(x,y)}{fillcolor}{n1}{n2}{n3}{n4}{n5}
% n_i=1 iff node i is in the subdiagram.
%
% White nodes have square -2.
% Black nodes have square -4.
%
% Integral root Gram matrix:
%
%   [ -2  1  1  0  0 ]
%   [  1 -2  1  0  0 ]
%   [  1  1 -2  2  0 ]
%   [  0  0  2 -4  4 ]
%   [  0  0  0  4 -4 ]
%
% Ambient diagram:
%
%       3 == 4 === 5
%      / \
%     1---2
%
% simple white-white edge:      (r_i,r_j)=1
% double white-black edge 3==4: (r_3,r_4)=2
% bold black-black edge 4===5:  (r_4,r_5)=4
% no edge:                      (r_i,r_j)=0

\newcommand{\SubdiagramCell}[8]{%
  \node[cell, fill=#3] (#1) at #2 {};
  \begin{scope}[shift={(#1.center)}, x=.55cm, y=.55cm]
    \coordinate (v1) at (-1.10,-0.35);
    \coordinate (v2) at (-0.25,-0.35);
    \coordinate (v3) at (-0.68,0.39);
    \coordinate (v4) at (0.20,0.92);
    \coordinate (v5) at (1.05,0.92);

    % Faded full ambient diagram.
    \draw[inactiveedge] (v1)--(v2);
    \draw[inactiveedge] (v1)--(v3);
    \draw[inactiveedge] (v2)--(v3);
    \draw[inactivedoubleedge] (v3)--(v4);
    \draw[inactiveboldedge] (v4)--(v5);

    \node[inactivewhite, circle, inner sep=0pt, minimum size=3.2mm] at (v1) {};
    \node[inactivewhite, circle, inner sep=0pt, minimum size=3.2mm] at (v2) {};
    \node[inactivewhite, circle, inner sep=0pt, minimum size=3.2mm] at (v3) {};
    \node[inactiveblack, circle, inner sep=0pt, minimum size=3.2mm] at (v4) {};
    \node[inactiveblack, circle, inner sep=0pt, minimum size=3.2mm] at (v5) {};

    % Exact selected-subdiagram glow: one capsule per selected edge,
    % and one halo per selected vertex. No hull is taken across unrelated
    % vertices or disconnected components.
    \ifnum#4=1\ifnum#5=1\GlowSegment{v1}{v2}\fi\fi
    \ifnum#4=1\ifnum#6=1\GlowSegment{v1}{v3}\fi\fi
    \ifnum#5=1\ifnum#6=1\GlowSegment{v2}{v3}\fi\fi
    \ifnum#6=1\ifnum#7=1\GlowSegment{v3}{v4}\fi\fi
    \ifnum#7=1\ifnum#8=1\GlowSegment{v4}{v5}\fi\fi

    \ifnum#4=1\GlowVertex{v1}\fi
    \ifnum#5=1\GlowVertex{v2}\fi
    \ifnum#6=1\GlowVertex{v3}\fi
    \ifnum#7=1\GlowVertex{v4}\fi
    \ifnum#8=1\GlowVertex{v5}\fi

    % Active selected edges.
    \ifnum#4=1\ifnum#5=1\draw[activeedge] (v1)--(v2);\fi\fi
    \ifnum#4=1\ifnum#6=1\draw[activeedge] (v1)--(v3);\fi\fi
    \ifnum#5=1\ifnum#6=1\draw[activeedge] (v2)--(v3);\fi\fi
    \ifnum#6=1\ifnum#7=1\draw[activedoubleedge] (v3)--(v4);\fi\fi
    \ifnum#7=1\ifnum#8=1\draw[activeboldedge] (v4)--(v5);\fi\fi

    % Active selected nodes, redrawn over the glow.
    \ifnum#4=1
      \node[activewhite, circle, inner sep=0pt, minimum size=3.2mm] at (v1) {};
    \fi
    \ifnum#5=1
      \node[activewhite, circle, inner sep=0pt, minimum size=3.2mm] at (v2) {};
    \fi
    \ifnum#6=1
      \node[activewhite, circle, inner sep=0pt, minimum size=3.2mm] at (v3) {};
    \fi
    \ifnum#7=1
      \node[activeblack, circle, inner sep=0pt, minimum size=3.2mm] at (v4) {};
    \fi
    \ifnum#8=1
      \node[activeblack, circle, inner sep=0pt, minimum size=3.2mm] at (v5) {};
    \fi
  \end{scope}
}
