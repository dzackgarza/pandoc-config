% Poset of subdiagrams of the Coxeter diagram C(L) of the lattice with root
% Gram matrix G_L below. White roots have square -2, black roots square -4:
%
%       3 == 4 === 5        1-2, 1-3, 2-3: (r_i, r_j) = 1   (weight 3)
%      / \                  3-4:           (r_3, r_4) = 2   (weight 4)
%     1---2                 4-5:           (r_4, r_5) = 4   (weight infinity)
%
% A subdiagram is a subset of {1,...,5}, encoded as the bitmask
% m = sum of 2^(i-1) over its vertices i. Cover arrows point from a subdiagram
% to the subdiagrams with one more vertex.
\begin{tikzpicture}[scale=0.72]
  % Cell positions: one row per rank, subsets in lexicographic order.
  \foreach \rank/\n/\row in {0/1/{0}, 1/5/{1, 2, 4, 8, 16},
                             2/10/{3, 5, 9, 17, 6, 10, 18, 12, 20, 24},
                             3/10/{7, 11, 19, 13, 21, 25, 14, 22, 26, 28},
                             4/5/{15, 23, 27, 29, 30}, 5/1/{31}}
    \foreach \m [count=\i] in \row
      \path (({(\i - (\n + 1) / 2) * 2.451}, {2.05 * \rank})
        node[minimum width=1.57cm, minimum height=1.07cm] (S\m) {};

  % Cover relations: add one vertex i not in m.
  \foreach \m in {0, ..., 30}
    \foreach \v in {1, ..., 5} {
      \pgfmathtruncatemacro\sel{mod(floor(\m / 2^(\v - 1)), 2)}
      \pgfmathtruncatemacro\up{\m + 2^(\v - 1)}
      \ifnum\sel=0
        \scoped[on background layer]
          \draw[degeneration, dzg muted] (S\m) -- (S\up);
      \fi
    }

  % Cells, by bitmask: the type of the subdiagram and the diagram, with the
  % vertices and edges outside the subdiagram inactive. Types computed from
  % G_L: elliptic when -G_L on the subset is positive definite; parabolic
  % when it is positive semidefinite with one kernel direction per connected
  % component (every component affine), which holds for {1,2,3} = A~_2 and
  % {4,5} = A~_1 only, and these two are maximal; other otherwise, including
  % A~_2 + A_1 and A~_1 + A_1, whose elliptic component is not affine.
  \foreach \type [count=\c from 0] in {elliptic, elliptic, elliptic, elliptic,
      elliptic, elliptic, elliptic, parabolic, elliptic, elliptic, elliptic,
      elliptic, elliptic, elliptic, elliptic, other, elliptic, elliptic,
      elliptic, elliptic, elliptic, elliptic, elliptic, other, parabolic,
      other, other, other, other, other, other, other} {
    \scoped[on background layer]
      \node[subdiagram cell=\type, minimum width=1.57cm, minimum height=1.07cm]
        at (S\c) {};
    \begin{scope}[shift={(S\c)}, x=0.55cm, y=0.55cm]
      \foreach \v/\mark/\pos in {1/white/{(-1.10, -0.35)}, 2/white/{(-0.25, -0.35)},
                                 3/white/{(-0.68, 0.39)}, 4/black/{(0.20, 0.92)},
                                 5/black/{(1.05, 0.92)}} {
        \pgfmathtruncatemacro\sel{mod(floor(\c / 2^(\v - 1)), 2)}
        \ifnum\sel=1
          \node[vertex=\mark] (v\v) at \pos {};
          \scoped[on background layer] \draw[parabolic] (v\v.center) -- (v\v.center);
        \else
          \node[inactive, vertex=\mark] (v\v) at \pos {};
        \fi
      }
      \foreach \a/\b/\w in {1/2/3, 1/3/3, 2/3/3, 3/4/4, 4/5/infinity} {
        \pgfmathtruncatemacro\sel{mod(floor(\c / 2^(\a - 1)), 2)
                                  * mod(floor(\c / 2^(\b - 1)), 2)}
        \ifnum\sel=1
          \scoped[on edge layer] \draw[coxeter edge=\w] (v\a.center) -- (v\b.center);
          \scoped[on background layer] \draw[parabolic] (v\a.center) -- (v\b.center);
        \else
          \scoped[on edge layer]
            \draw[inactive, coxeter edge=\w] (v\a.center) -- (v\b.center);
        \fi
      }
    \end{scope}
  }

  \node[font=\small] at (0, 11.45) {Subdiagram Poset of $C(L)$};

  \foreach \type/\name [count=\i] in {elliptic/Elliptic,
                                     parabolic/Maximal Parabolic, other/Neither}
    \node[subdiagram cell=\type, minimum width=0.47cm, minimum height=0.27cm,
          label={[font=\scriptsize]right:\name}] at (7.9, {9.8 - 0.55 * \i}) {};

  \node at (-9, 0) {$G_L =
    \begin{pmatrix}
      -2 &  1 &  1 &  0 &  0 \\
       1 & -2 &  1 &  0 &  0 \\
       1 &  1 & -2 &  2 &  0 \\
       0 &  0 &  2 & -4 &  4 \\
       0 &  0 &  0 &  4 & -4
    \end{pmatrix}$};
\end{tikzpicture}
