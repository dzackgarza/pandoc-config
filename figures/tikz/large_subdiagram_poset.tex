% Poset of subdiagrams of the Coxeter diagram C(L) of the lattice with root
% Gram matrix G_L: objects/posets/subdiagrams-C-L.json, laid out by
% \posetfromjson. The data file's script gives C(L), G_L and the type of each
% subdiagram (elliptic, parabolic, other), which is the style of its cell. An
% element is named by the bitmask m = sum of 2^(i-1) over its vertices i;
% cover arrows point from a subdiagram to the subdiagrams with one more
% vertex. Each cell shows C(L) with the vertices and edges outside the
% subdiagram inactive. The cells lie on the Dynkin behind layer, above the
% covers (background) and below the diagrams' edges, with the highlights of
% the subdiagrams drawn on them.
\begin{tikzpicture}
  \posetfromjson[name prefix=S-, grade distance=1.48cm, element width=1.57cm,
    element height=1.07cm, element sep=2mm, element positions=even,
    element layer=Dynkin behind,
    cover layer=background,
    element labels=false, cover style={degeneration, dzg muted},
    cover direction=up]{objects/posets/subdiagrams-C-L.json}
  \foreach \c in \posetelements {
    \begin{scope}[shift={(S-\c)}, x=0.396cm, y=0.396cm]
      \foreach \v/\mark/\pos in {1/white/{(-1.10, -0.35)}, 2/white/{(-0.25, -0.35)},
                                 3/white/{(-0.68, 0.39)}, 4/black/{(0.20, 0.92)},
                                 5/black/{(1.05, 0.92)}} {
        \pgfmathtruncatemacro\sel{mod(floor(\c / 2^(\v - 1)), 2)}
        \ifnum\sel=1
          \node[vertex=\mark] (v\v) at \pos {};
          \begin{pgfonlayer}{Dynkin behind}
            \draw[parabolic] (v\v.center) -- (v\v.center);
          \end{pgfonlayer}
        \else
          \node[inactive, vertex=\mark] (v\v) at \pos {};
        \fi
      }
      \foreach \a/\b/\w in {1/2/3, 1/3/3, 2/3/3, 3/4/4, 4/5/infinity} {
        \pgfmathtruncatemacro\sel{mod(floor(\c / 2^(\a - 1)), 2)
                                  * mod(floor(\c / 2^(\b - 1)), 2)}
        \ifnum\sel=1
          \scoped[on edge layer] \draw[coxeter edge=\w] (v\a.center) -- (v\b.center);
          \begin{pgfonlayer}{Dynkin behind}
            \draw[parabolic] (v\a.center) -- (v\b.center);
          \end{pgfonlayer}
        \else
          \scoped[on edge layer]
            \draw[inactive, coxeter edge=\w] (v\a.center) -- (v\b.center);
        \fi
      }
    \end{scope}}

  \node[font=\small, above=4mm of S-31] {Subdiagram Poset of $C(L)$};
  \foreach \type/\name [count=\i] in {elliptic/Elliptic,
                                     parabolic/Maximal Parabolic, other/Neither}
    \node[subdiagram cell=\type, minimum width=0.47cm, minimum height=0.27cm,
          label={[font=\scriptsize]right:\name}]
      at ([xshift=-2.5cm, yshift={-0.4cm * \i}]current bounding box.north east) {};
  \node[anchor=east] at ([xshift=-3mm]current bounding box.west |- S-0) {$G_L =
    \begin{pmatrix}
      -2 &  1 &  1 &  0 &  0 \\
       1 & -2 &  1 &  0 &  0 \\
       1 &  1 & -2 &  2 &  0 \\
       0 &  0 &  2 & -4 &  4 \\
       0 &  0 &  0 &  4 & -4
    \end{pmatrix}$};
\end{tikzpicture}
