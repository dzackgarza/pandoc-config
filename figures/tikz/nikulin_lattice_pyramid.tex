  \draw[lattice grid] (0,0) grid (22,12);
  \draw[->] (0,0) -- (22.1,0) node[right] {$r$};
  \draw[->] (0,0) -- (0,12) node[above] {$a$};
  \draw[<->] (0.5,0.5) -- (11,11) -- (21.15,0.85);

    \draw (\r,0.12) -- (\r,-0.12);
    \node[below=2pt, font=\scriptsize] at (\r,0) {\r};
  }
  \foreach \a in {1,...,11} {
    \draw (-0.12,\a) -- (0.12,\a);
    \node[left=4pt, font=\scriptsize] at (0,\a) {\a};
  }
  \foreach \r/\a/\m in \nikulinTwoElementaryPoints
    \node[vertex=\m] at (\r,\a) {};
  \foreach \m/\text [count=\i] in
      {black/{$\delta=1$ only}, white/{$\delta=0$ only}, doubled/both}
    \node[vertex=\m, label={[label distance=0.3cm]right:\text}]
      at (17.2, 11.7-0.9*\i) {};
