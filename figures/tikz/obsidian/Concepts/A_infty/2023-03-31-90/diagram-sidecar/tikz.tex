\documentclass[tikz,border=3pt]{standalone}

% Core libraries (subset of preamble that is actually needed)
\usetikzlibrary{
    arrows.meta,
    calc,
    positioning,
}

\tikzset{>=Stealth}
\tikzset{
    mathnode/.style={rectangle, draw=none, fill=none, inner sep=2pt},
    labelnode/.style={rectangle, draw=none, fill=none, font=\small, inner sep=1pt},
}

\begin{document}

\begin{tikzpicture}[
    every node/.style={mathnode},
    dot/.style={circle, fill=black, inner sep=2pt},
    smalldot/.style={circle, fill=black, inner sep=1.5pt},
]

% ================================================================
%  Crop 1: Corolla Tree (planar rooted tree)
% ================================================================
\begin{scope}[shift={(-6,0)}]
  % --- Caption ---
  \node[anchor=south, font=\bfseries] at (0,5.5)
    {$b_n$ as a corolla tree};

  % --- Central vertex: the operation itself (m_n / b_n) ---
  \node[dot] (v) at (0,0) {};
  % Small label identifying the operation vertex
  \node[above right=1mm of v, font=\small] {$m_n$};

  % --- Input edges: 8 strands fanning upward ---
  \foreach \i [evaluate={\a=160-(\i-1)*20}] in {1,...,8} {
    \node[smalldot] (l\i) at (\a:3.5) {};
    \draw (v) -- (l\i);
  }

  % --- Input labels ---
  \node at (160:4.3) {$a_1$};
  \node at (140:4.3) {$a_2$};
  \node at (120:4.3) {$a_3$};
  % Ellipsis for unlabelled intermediate inputs (a_4 through a_{n-1})
  \node at (80:4.5) {$\cdots$};
  \node at (20:4.3) {$a_n$};

  % --- Output edge (descending) ---
  \node[smalldot] (out) at (0,-2.8) {};
  \draw (v) -- (out);
  \node at (0,-3.6) {$b_n$};

  % --- Annotation: arity-n operation ---
  \node[anchor=north west, font=\scriptsize, gray, align=left] at (-3, -4)
    {Note: central vertex = operation $m_n$,\\[2pt]
     output $b_n$ at bottom of root edge.};
\end{scope}


% ================================================================
%  Crop 2: Halfdisk (subdivided upper arc, single base chord)
% ================================================================
\begin{scope}[shift={(6,0)}]
  % --- Caption ---
  \node[anchor=south, font=\bfseries] at (0,5.5)
    {$b_n$ as a halfdisk};

  % --- 10 vertices on the upper semicircle ---
  \foreach \i [evaluate={\a=180-(\i-1)*20}] in {1,...,10} {
    \node[smalldot] (p\i) at (\a:3) {};
  }

  % --- 9 edges along the upper arc ---
  \foreach \i [evaluate=\i as \next using int(\i+1)] in {1,...,9} {
    \draw (p\i) -- (p\next);
  }

  % --- 1 base edge (output) ---
  \draw[thick] (p1) -- (p10);

  % --- Labels on upper-arc edges (first three) ---
  \path (p1) -- (p2) node[midway, above=3mm] {$a_1$};
  \path (p2) -- (p3) node[midway, above=3mm] {$a_2$};
  \path (p3) -- (p4) node[midway, above=3mm] {$a_3$};

  % Ellipsis for the remaining upper-arc edges
  \path (p5) -- (p6) node[midway, above=3mm] {$\cdots$};
  \path (p6) -- (p7) node[midway, above=3mm] {$a_n$};

  % --- Output label on base edge ---
  \path (p1) -- (p10) node[midway, below=3mm] {$b_n$};

  % --- Annotation ---
  \node[anchor=north, font=\scriptsize, gray, align=center] at (0, -4)
    {10 vertices, 9 upper-arc edges (inputs),\\[2pt]
     1 base edge (output $b_n$).};
\end{scope}

\end{tikzpicture}

\end{document}
