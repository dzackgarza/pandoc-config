% The A-infinity operation b_n drawn two ways: as a corolla tree (inputs
% a_1..a_n on the leaves, output b_n on the root) and as a halfdisk (inputs on
% the subdivided upper arc, output on the base chord).
\documentclass[tikz,border=3pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}
% Corolla tree
\begin{scope}[shift={(-6,0)}]
  \node[anchor=south, font=\bfseries] at (0,5.5) {$b_n$ as a corolla tree};

  \node[dual vertex, label={[font=\small]above right:$m_n$}] (v) at (0,0) {};
  % Eight input leaves at angles 160, 140, ..., 20.
  \foreach \i [evaluate={\a=160-(\i-1)*20}] in {1,...,8} {
    \node[curve node] (l\i) at (\a:3.5) {};
    \draw (v) -- (l\i);
  }
  \foreach \a/\lbl in {160/a_1, 140/a_2, 120/a_3, 20/a_n}
    \node at (\a:4.3) {$\lbl$};
  \node at (80:4.5) {$\cdots$};

  \node[curve node, label={below:$b_n$}] (out) at (0,-2.8) {};
  \draw (v) -- (out);

  \node[anchor=north west, font=\scriptsize, text=dzg muted, align=left] at (-3,-4)
    {Note: central vertex = operation $m_n$,\\[2pt]
     output $b_n$ at bottom of root edge.};
\end{scope}

% Halfdisk
\begin{scope}[shift={(6,0)}]
  \node[anchor=south, font=\bfseries] at (0,5.5) {$b_n$ as a halfdisk};

  % Ten vertices on the upper semicircle, joined in order by nine edges.
  \foreach \i [evaluate={\a=180-(\i-1)*20}] in {1,...,10}
    \node[curve node] (p\i) at (\a:3) {};
  \foreach \i [evaluate={\next=int(\i+1)}] in {1,...,9}
    \draw (p\i) -- (p\next);
  \draw (p1) -- node[below=3mm] {$b_n$} (p10);

  \foreach \i/\j/\lbl in {1/2/a_1, 2/3/a_2, 3/4/a_3, 5/6/\cdots, 6/7/a_n}
    \path (p\i) -- node[above=3mm] {$\lbl$} (p\j);

  \node[anchor=north, font=\scriptsize, text=dzg muted, align=center] at (0,-4)
    {10 vertices, 9 upper-arc edges (inputs),\\[2pt]
     1 base edge (output $b_n$).};
\end{scope}
\end{tikzpicture}
\end{document}
