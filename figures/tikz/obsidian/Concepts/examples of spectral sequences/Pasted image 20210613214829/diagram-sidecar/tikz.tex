% E^2 page of the homological Serre spectral sequence of a fibration
% F → E → B with F ≅ S^1: two rows (q = 0, 1), five d^2 differentials
% (p,0) → (p-2,1) for p = 2..6.
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}
  \node (E2) {
    $\begin{tikzcd}[row sep=4em, column sep=3.5em]
      % q = 1
        \mathbb{Z} & H_1(B) & H_2(B) & H_3(B) & H_4(B) & H_5(B) & H_6(B) & \cdots
      \\
      % q = 0
        \mathbb{Z}
      & H_1(B)
      & H_2(B) \arrow[to=1-1]
      & H_3(B) \arrow[to=1-2]
      & H_4(B) \arrow[to=1-3]
      & H_5(B) \arrow[to=1-4]
      & H_6(B) \arrow[to=1-5]
      & \cdots
    \end{tikzcd}$
  };
  % L-shaped axes around the page.
  \node[fit=(E2), inner xsep=0.8em, inner ysep=1.2em] (page) {};
  \draw (page.north west) -- (page.south west) -- (page.south east);
  \node[below, font=\small] at (page.south east) {$p$ (base degree)};
  \node[left, font=\small] at (page.north west) {$q$ (fiber degree)};
\end{tikzpicture}
\end{document}
