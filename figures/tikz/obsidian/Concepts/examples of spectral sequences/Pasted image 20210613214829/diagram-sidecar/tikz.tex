% E^2 page of the homological Serre spectral sequence of a fibration
% F → E → B with F ≅ S^1: two rows (q = 0, 1), five d^2 differentials
% (p,0) → (p-2,1) for p = 2..6.
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}
  \pic (E) [ss homological, ss p={0,...,7}, ss q={0,1}, ss ticks=false,
    ss p name={$p$ (base degree)}, ss q name={$q$ (fiber degree)},
    ss column sep=1.8cm, ss row sep=1.5cm] {ss page};
  \foreach \q in {0,1} {
    \ssentry{E}{0}{\q}{$\mathbb{Z}$}
    \foreach \p in {1,...,6} {\ssentry{E}{\p}{\q}{$H_{\p}(B)$}}
    \ssentry{E}{7}{\q}{$\cdots$}}
  \foreach \p in {2,...,6} {\ssdifferential[ss unlabelled]{E}{2}{\p}{0}}
\end{tikzpicture}
\end{document}
