%

%   φ_{23} = id_B ⊗ φ   : BBN → BNB  (top edge)

%   φ_{13} = φ_{12} ∘ φ_{23} : BBN → NBB  (left slanted edge)
%
% Cocycle condition: φ_{13} = φ_{12} ∘ φ_{23}
%

    \arrow[r, "\phi_{23}"]

    \arrow[d, "\phi_{12}"]
\\
&
N \otimes_A B \otimes_A B
\end{tikzcd}
