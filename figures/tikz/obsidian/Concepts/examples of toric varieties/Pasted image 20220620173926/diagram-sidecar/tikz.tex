% Moment polytope P of the weighted projective space P(1,1,2):
% the lattice triangle with vertices v0 = (0,0), v1 = (2,0), v2 = (0,1),
% with the lattice point (1,0) on its bottom edge.
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=3.5]
  \path[polygon region] (0,0) -- (2,0) -- (0,1) -- cycle;
  \draw[polytope edge] (0,0) -- (2,0) -- node[sloped, above] {$P$} (0,1) -- cycle;
  \foreach \p/\pos/\lbl in {(0,0)/below left/v_0, (2,0)/below right/v_1,
      (0,1)/above left/v_2, (1,0)/below/{}}
    \node[distinguished point, label={\pos:$\lbl$}] at \p {};
\end{tikzpicture}
\end{document}
