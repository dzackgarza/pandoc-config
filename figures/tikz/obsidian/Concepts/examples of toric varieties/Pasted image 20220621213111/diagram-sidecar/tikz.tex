% A fan with two hatched maximal cones: σ0 = Cone(e1, e2) and
% σ1 = Cone(e1, 2e1 - e2), over the lattice points in [0,3] x [0,2].
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=1.8]
  \foreach \x in {0,...,3}
    \foreach \y in {0,...,2}
      \node[lattice point] at (\x,\y) {};

  \fill[hatched region] (0,0) -- (3,0) -- (0,3) -- cycle;
  \fill[hatched region] (0,0) -- (3,0) -- (4,-2) -- cycle;

  \draw[fan ray] (0,0) -- (3,0) node[below] {$e_1$};
  \draw[fan ray] (0,0) -- (0,2.5) node[left] {$e_2$};
  \draw[fan ray] (0,0) -- (4,-2) node[below right] {$2e_1 - e_2$};
  \node[below left] at (0,0) {$0$};
\end{tikzpicture}
\end{document}
