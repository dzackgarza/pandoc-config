% The fan of P^2: rays v1 = (1,0), v2 = (0,1), v3 = (-1,-1) and the three
% maximal cones σ0 = Cone(v1,v2), σ1 = Cone(v2,v3), σ2 = Cone(v3,v1).
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=2.5]
  \path[polygon region] (0,0) -- (0:1.3) arc[start angle=0, end angle=90, radius=1.3] -- cycle;
  \path[fill=dzg blob] (0,0) -- (90:1.3) arc[start angle=90, end angle=225, radius=1.3] -- cycle;
  \path[fill=dzg grid!40] (0,0) -- (225:1.3) arc[start angle=225, end angle=360, radius=1.3] -- cycle;

  \draw[fan ray] (0,0) -- (1.3,0) node[right] {$v_1$};
  \draw[fan ray] (0,0) -- (0,1.3) node[above] {$v_2$};
  \draw[fan ray] (0,0) -- (-1.3,-1.3) node[below left] {$v_3$};
  \node[distinguished point, label={below left:$0$}] at (0,0) {};

  \foreach \p/\lbl in {(0.55,0.55)/\sigma_0, (-0.55,0.50)/\sigma_1, (0.45,-0.45)/\sigma_2}
    \node[font=\small] at \p {$\lbl$};
\end{tikzpicture}
\end{document}
