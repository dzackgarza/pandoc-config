% A toric surface X_Δ (left, a schematic three-lobed surface over x, y, z
% axes) and its lattice polytope Δ, the triangle with vertices (0,0), (2,1),
% (1,2) and interior point (1,1) (right), linked by the moment map.
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=0.7, font=\sffamily, use Hobby shortcut]
% Toric surface.
\begin{scope}[shift={(-8,0)}, scale=1.1]
  % Back of the surface: back lobe, its wing, and the cavity underneath.
  \filldraw[fill=dzg blob, draw=dzg muted, closed=true]
    (0,0.2) .. (-1.2,0.35) .. (-2.2,-0.5) .. (-2.1,-1.5) .. (-1.6,-1.4)
    .. (-0.6,-0.8) .. (0,-0.2);
  \filldraw[fill=dzg blob, draw=dzg muted, closed=true]
    (0,0.2) .. (-0.9,0.8) .. (-1.8,0.6) .. (-2.2,-0.5) .. (-1,-0.1);
  \filldraw[fill=dzg grid, draw=dzg muted, closed=true]
    (0,-0.3) .. (1.0,-1.6) .. (-0.4,-2.0) .. (-1.6,-1.4) .. (-0.7,-0.7);

  % Front of the surface: two lobes and the central saddle.
  \filldraw[fill=dzg region, draw=dzg accent, closed=true]
    (0,0.3) .. (1.2,0.9) .. (2.4,0.6) .. (2.7,-0.1) .. (2.4,-0.8)
    .. (1.4,-1.0) .. (0.8,-0.6) .. (0,-0.1);
  \filldraw[fill=dzg region, draw=dzg accent, closed=true]
    (0,0.3) .. (-1.0,1.3) .. (-2.0,1.6) .. (-2.6,1.0) .. (-2.4,0.4)
    .. (-1.4,-0.1) .. (-0.8,0) .. (0,0.1);
  \filldraw[fill=dzg region, draw=dzg accent, closed=true]
    (0,0.4) .. (1.4,0.2) .. (0,0) .. (-1.4,0.2);

  % Axes.
  \draw[coordinate axis] (-0.5,-2.8) -- (2.5,-2.8) node[right, font=\small] {$x$};
  \draw[coordinate axis] (-0.5,-2.8) -- (-0.5,0) node[above, font=\small] {$y$};
  \draw[coordinate axis, dashed] (-0.5,-2.8) -- (1.5,-3.8) node[below right, font=\small] {$z$};
  \foreach \x in {0,1,2}
    \draw[dzg muted] (\x-0.5,-2.85) -- (\x-0.5,-2.75);
  \foreach \y in {-2,-1}
    \draw[dzg muted] (-0.55,\y) -- (-0.45,\y);
  \node[below left, font=\small] at (-0.5,-2.8) {$0$};

  \node[font=\small\itshape, above] at (0.5,2.2) {Toric surface $X_\Delta$};
\end{scope}

% Lattice polytope.
\begin{scope}[shift={(3,-1)}, scale=0.9]
  \foreach \x in {-1,...,5}
    \foreach \y in {-1,...,5}
      \node[lattice point] at (\x,\y) {};

  \path[polygon region] (0,0) -- (2,1) -- (1,2) -- cycle;
  \draw[polytope edge] (0,0) -- (2,1) -- (1,2) -- cycle;

  % Rays of the normal fan through the vertices (2,1) and (1,2).
  \draw[dashed, dzg muted] (0,0) -- (2.8,1.4);
  \draw[dashed, dzg muted] (0,0) -- (1.4,2.8);

  \draw[coordinate axis] (-0.3,0) -- (5.2,0) node[right, font=\small] {$m_1$};
  \draw[coordinate axis] (0,-0.3) -- (0,5.2) node[above, font=\small] {$m_2$};
  \foreach \t in {0,...,4} {
    \node[below, font=\small] at (\t,0) {$\t$};
    \node[left, font=\small] at (0,\t) {$\t$};
  }

  \foreach \p/\pos/\lbl in {(0,0)/below left/{(0,0)}, (2,1)/below right/{(2,1)},
      (1,2)/above/{(1,2)}}
    \node[distinguished point, label={[font=\small]\pos:$\lbl$}] at \p {};
  \node[distinguished point, fill=dzg singular,
    label={[font=\small]above right:$(1,1)$}] at (1,1) {};

  \node[font=\small\itshape, above] at (2.5,5.5) {Lattice polytope $\Delta$};
\end{scope}

\draw[->] (-3.5,1.5) -- node[above=0.3em, font=\small\itshape]
  {moment map / normal fan} (-1.5,1.5);
\end{tikzpicture}
\end{document}
