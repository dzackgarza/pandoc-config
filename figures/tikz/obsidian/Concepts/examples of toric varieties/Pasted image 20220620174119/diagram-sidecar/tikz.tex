% The lattice quadrilateral P = P_{2,4} and its normal fan Σ_P
% (Cox-Little-Schenck, Toric Varieties, Example 2.3.16).
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=0.7, every label/.append style={font=\small}]
% Lattice polytope P_{2,4}.
\begin{scope}
  \path[polygon region] (0,0) -- (0,1) -- (4,1) -- (2,0) -- cycle;
  \draw[polytope edge] (0,0) -- (0,1) -- (4,1) -- (2,0) -- cycle;
  \foreach \p/\pos/\lbl in {(0,0)/below left/v_1, (0,1)/above left/v_2,
      (4,1)/above right/v_3, (2,0)/below right/v_4}
    \node[distinguished point, label={\pos:$\lbl$}] at \p {};
  \foreach \p in {(1,0), (1,1), (2,1), (3,1)}
    \node[lattice point] at \p {};
  \node[font=\small] at (1.55,0.45) {$P = P_{2,4}$};
\end{scope}

% Normal fan Σ_P: inward normals of the four edges.
\begin{scope}[xshift=7.5cm]
  \path[polygon region] (0,0) -- (2.5,0) -- (0,2.5) -- cycle;
  \path[polygon region] (0,0) -- (2.5,0) -- (0,-2.5) -- cycle;
  \path[polygon region] (0,0) -- (0,-2.5) -- (-1.5,3) -- cycle;
  \path[polygon region] (0,0) -- (-1.5,3) -- (0,2.5) -- cycle;

  \draw[fan ray] (0,0) -- (0,2.5)  node[above, font=\small] {$\rho_1 = (0,1)$};
  \draw[fan ray] (0,0) -- (2.5,0)  node[right, font=\small] {$\rho_2 = (1,0)$};
  \draw[fan ray] (0,0) -- (0,-2.5) node[below, font=\small] {$\rho_3 = (0,-1)$};
  \draw[fan ray] (0,0) -- (-1.5,3) node[above left, font=\small] {$\rho_4 = (-1,2)$};
  \node[distinguished point] at (0,0) {};

  \foreach \p/\lbl in {(1.4,1.4)/\sigma_1, (1.4,-1.4)/\sigma_2, (-0.7,-0.7)/\sigma_3,
      (-0.6,1.0)/\sigma_4, (-1.0,2.0)/\Sigma_P}
    \node[font=\small] at \p {$\lbl$};
\end{scope}
\end{tikzpicture}
\end{document}
