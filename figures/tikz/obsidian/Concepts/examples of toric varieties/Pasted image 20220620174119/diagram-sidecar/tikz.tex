% The lattice quadrilateral P = P_{2,4} and its normal fan Σ_P
% (Cox-Little-Schenck, Toric Varieties, Example 2.3.16).
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=0.7, every label/.append style={font=\small}]
% Lattice polytope P_{2,4}.
\begin{scope}
  \path[polygon region] (0,0) -- (0,1) -- (4,1) -- (2,0) -- cycle;
  \draw[polytope edge] (0,0) -- (0,1) -- (4,1) -- (2,0) -- cycle;
  \foreach \p/\pos/\lbl in {(0,0)/below left/v_1, (0,1)/above left/v_2,
      (4,1)/above right/v_3, (2,0)/below right/v_4}
    \node[distinguished point, label={\pos:$\lbl$}] at \p {};
  \foreach \p in {(1,0), (1,1), (2,1), (3,1)}
    \node[lattice point] at \p {};
  \node[font=\small] at (1.55,0.45) {$P = P_{2,4}$};
\end{scope}

% Normal fan Σ_P, the fan of F_2: inward normals of the four edges.
\pic (S) at (8.5,0) {object=toric/fan-F2};
\node[font=\small] at ($(S-0)+(-2,-1.8)$) {$\Sigma_P$};
\end{tikzpicture}
\end{document}
