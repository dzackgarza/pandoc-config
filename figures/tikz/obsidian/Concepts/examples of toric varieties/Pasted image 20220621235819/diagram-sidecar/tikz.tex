% The strongly convex cone σ = Cone(e1, e2, e3) in N_R = R^3, seen from above
% the first octant, with the planes z = 0 and y = 0 for reference. The cone is
% shaded by its section with the plane x + y + z = 4. Tick labels are the
% source's values (five per unit of the picture).
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[x={(3.0cm,1.2cm)}, y={(2.0cm,-1.5cm)}, z={(0cm,3.0cm)}]
  % Reference planes z = 0 and y = 0.
  \filldraw[fill=dzg grid, fill opacity=0.3, draw=dzg grid]
    (-4.5,-4.5,0) -- (4.8,-4.5,0) -- (4.8,4.8,0) -- (-4.5,4.8,0) -- cycle;
  \filldraw[fill=dzg region, fill opacity=0.8, draw=dzg grid]
    (-4.5,0,-4.5) -- (4.8,0,-4.5) -- (4.8,0,4.8) -- (-4.5,0,4.8) -- cycle;

  % Axes; the parts hidden behind the planes are dashed.
  \draw[coordinate axis, dashed] (0,0,0) -- (4.8,0,0) node[anchor=south west] {$x$};
  \draw[coordinate axis] (0,0,0) -- (-4.5,0,0);
  \draw[coordinate axis] (0,0,0) -- (0,4.8,0) node[anchor=south east] {$y$};
  \draw[coordinate axis] (0,0,0) -- (0,-4.5,0);
  \draw[coordinate axis] (0,0,0) -- (0,0,4.8) node[anchor=south] {$z$};
  \draw[coordinate axis, dashed] (0,0,0) -- (0,0,-4.5);

  % Ticks at every fifth unit of the source.
  \foreach \t [evaluate={\v=int(5*\t)}] in {-5,-4,-3,-2,-1,1,2,3,4,5,6} {
    \draw[dzg muted] (0.8*\t,0.12,0) -- (0.8*\t,-0.12,0);
    \node[anchor=north, font=\tiny, inner sep=1pt] at (0.8*\t,-0.2,0) {$\v$};
  }
  \foreach \t [evaluate={\v=int(5*\t)}] in {-4,-3,-2,-1,1,2,3,4,5,6} {
    \draw[dzg muted] (0.12,0.8*\t,0) -- (-0.12,0.8*\t,0);
    \node[anchor=east, font=\tiny, inner sep=1pt] at (-0.25,0.8*\t,0) {$\v$};
  }
  \foreach \t [evaluate={\v=int(5*\t)}] in {-2,-1,1,2} {
    \draw[dzg muted] (0.08,0,0.8*\t) -- (-0.08,0,0.8*\t);
    \node[anchor=south east, font=\tiny, inner sep=1pt] at (-0.1,0,0.8*\t) {$\v$};
  }

  % The cone: its section with x + y + z = 4, and its three rays.
  \filldraw[fill=dzg accent, fill opacity=0.3, draw=dzg accent]
    (4,0,0) -- (0,4,0) -- (0,0,4) -- cycle;
  \draw[fan ray, dashed] (0,0,0) -- (4,0,0);
  \draw[fan ray] (0,0,0) -- (0,4,0);
  \draw[fan ray] (0,0,0) -- (0,0,4);

  \node[anchor=north east, font=\small] at (0,0,0) {$0$};
\end{tikzpicture}
\end{document}
