% Fan of the blowup of A^2 at the origin: rays v1 = (0,1), v2 = (1,1),
% v3 = (1,0), maximal cones σ0 = Cone(v1,v2) and σ1 = Cone(v2,v3). The ray
% v2 of the exceptional divisor is dashed.
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=1.3]
  \path[polygon region] (0,0) -- (0,3.3) -- (3.3,3.3) -- cycle;
  \path[fill=dzg blob] (0,0) -- (3.3,0) -- (3.3,3.3) -- cycle;
  \draw[lattice grid] (0,0) grid (3.5,3.5);
  \foreach \x in {0,...,3}
    \foreach \y in {0,...,3}
      \node[lattice point] at (\x,\y) {};

  \draw[fan ray] (0,0) -- (0,3.3);
  \draw[fan ray] (0,0) -- (3.3,0);
  \draw[fan ray, dashed] (0,0) -- (3.3,3.3);

  \node[font=\small, right] at (0.3,2.8) {$v_1 = (0,1)$};
  \node[font=\small, above] at (2.8,0.3) {$v_3 = (1,0)$};
  \node[font=\small, above right] at (2.8,2.8) {$v_2 = (1,1)$};
  \node at (0.7,1.8) {$\sigma_0$};
  \node at (1.8,0.7) {$\sigma_1$};
  \node[font=\small, below left] at (0,0) {$0$};

  \node[draw, font=\small, align=left, anchor=north west] at (3.8,3.5) {
    $v_1 = (0,1)$\\
    $v_2 = (1,1)$ (exceptional)\\
    $v_3 = (1,0)$\\
    $\sigma_0 = \mathrm{Cone}(v_1,v_2)$\\
    $\sigma_1 = \mathrm{Cone}(v_2,v_3)$
  };
\end{tikzpicture}
\end{document}
