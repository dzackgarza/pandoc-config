% The strongly convex rational polyhedral cone σ = Cone(v1, v2), v1 = (0,1),
% v2 = (3,-1) (Cox-Little-Schenck, Toric Varieties, Section 2.2, m = 3):
% σ = {x >= 0, y >= -x/3}, hatched up to the edge of the picture.
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=0.85]
  \draw[coordinate axis] (-0.5,0) -- (6.5,0) node[right, font=\small] {$x$};
  \draw[coordinate axis] (0,-0.5) -- (0,6.5) node[above, font=\small] {$y$};
  \foreach \t in {1,...,5} {
    \node[below, font=\tiny, text=dzg muted] at (\t,0) {$\t$};
    \node[left, font=\tiny, text=dzg muted] at (0,\t) {$\t$};
  }

  \path[polygon region] (0,0) -- (0,5.5) -- (6,5.5) -- (6,-2) -- cycle;
  \fill[hatched region] (0,0) -- (0,5.5) -- (6,5.5) -- (6,-2) -- cycle;

  \foreach \x in {0,...,5}
    \foreach \y in {0,...,5}
      \node[lattice point] at (\x,\y) {};

  \draw[fan ray] (0,0) -- (0,5.5) node[above left, font=\small] {$v_1$};
  \draw[fan ray] (0,0) -- (6,-2) node[right, font=\small] {$v_2$};
  \node[below left, font=\small] at (0,0) {$0$};
  \node[font=\small] at (5.5,5.5) {$\sigma$};
\end{tikzpicture}
\end{document}
