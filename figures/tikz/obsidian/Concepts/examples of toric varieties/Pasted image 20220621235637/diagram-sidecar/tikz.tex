% The singular two-dimensional cone σ = Cone(v1, v2), v1 = (0,1), v2 = (2,-1):
% det(v1, v2) = -2, so the affine toric surface has an A1 singularity.
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=1.2]
  % σ = {x >= 0, y >= -x/2}, cut off at the edge of the grid.
  \path[polygon region] (0,0) -- (0,2.5) -- (4.5,2.5) -- (4.5,-2.25) -- cycle;
  \draw[lattice grid] (-0.5,-2.5) grid (4.5,2.5);
  \draw[coordinate axis] (-0.5,0) -- (4.5,0) node[below] {$x$};
  \draw[coordinate axis] (0,-2.5) -- (0,2.5) node[left] {$y$};
  \draw[fan ray] (0,0) -- (0,2.3) node[above] {$\rho_1$};
  \draw[fan ray] (0,0) -- (3.8,-1.9) node[right] {$\rho_2$};

  \foreach \p/\pos/\lbl in {(0,0)/below left/{(0,0)}, (1,0)/below/{(1,0)},
      (2,0)/below/{(2,0)}, (2,-1)/below/{(2,-1)}}
    \node[distinguished point, label={\pos:$\lbl$}] at \p {};
  \foreach \p in {(0,1), (0,2), (1,1)}
    \node[distinguished point] at \p {};

  \node[text=dzg accent, left] at (0.15,0.5) {$v_1 = (0,1)$};
  \node[text=dzg accent, below] at (1.0,-1.0) {$v_2 = (2,-1)$};
  \node[text=dzg accent, font=\small] at (3.5,-0.5)
    {$\sigma = \{(x,y)\mid x\ge0,\; y\ge -\frac{x}{2}\}$};
  \node[below left=2pt] at (0,0) {$O$};

  \node[draw=dzg singular, text=dzg singular, fill=white, rounded corners=2pt,
      align=left] at (1.8,1.8)
    {$\det(v_1, v_2) = -2$ \\ $\Rightarrow$ singular \\ $A_1$ singularity \\
     $\mathbb{C}^2 / \mathbb{Z}_2$};
\end{tikzpicture}
\end{document}
