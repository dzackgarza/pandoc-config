% The dual cones σ_i^∨ in M_R of the fan of P^2. The lines x = 0, y = 0 and
% y = -x cut the plane into six sectors; the three dual cones are hatched.
\documentclass[tikz,border=2pt]{standalone}
\usepackage{dzg-tikz}

\begin{document}
\begin{tikzpicture}[scale=2.5]
  \foreach \from/\to in {0/90, 135/180, 270/315}
    \fill[hatched region] (0,0) -- (\from:2.6)
      arc[start angle=\from, end angle=\to, radius=2.6] -- cycle;

  \foreach \a in {0, 90, 135}
    \draw[<->] (\a:2.8) -- (\a+180:2.8);

  \node[scale=1.1] at (1.0,0.6) {$\sigma_0^\vee$};
  \node[scale=1.1] at (-1.2,0.7) {$\sigma_1^\vee$};
  \node[scale=1.1] at (1.2,-0.7) {$\sigma_2^\vee$};
\end{tikzpicture}
\end{document}
