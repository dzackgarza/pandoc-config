% Commutative square: extension of morphisms to closures
% Context: Grothendieck's algebraic de Rham cohomology,
% resolution of singularities, compactification strategy.
%
% Nodes:
%   TL: X'  — pre-image of X under resolution map
%   TR: \overline{X'}  — smooth proper resolution of \overline{X}
%   BL: X  — original smooth variety
%   BR: \overline{X}  — compactification/closure of X
%
% Arrows:
%   Top horizontal: X' \hookrightarrow \overline{X'} (inclusion)
%   Bottom horizontal: X \hookrightarrow \overline{X} (inclusion)
%   Left vertical: X' \to X (isomorphism, per extraction context)
%   Right vertical: \overline{X'} \to \overline{X} (resolution map)
%
% The square commutes.

\begin{tikzcd}
X' \arrow[r, hook] \arrow[d, "\cong"'] & \overline{X'} \arrow[d] \\
X \arrow[r, hook] & \overline{X}
\end{tikzcd}
