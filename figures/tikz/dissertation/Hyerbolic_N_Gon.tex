% The ideal regular 18-gon in the Poincare disk, with vertices P_0..P_17.
% Needs LuaLaTeX (luahyperbolic). The picture is clipped to the disk, so the
% labels sit inside it.
\begin{luacode*}
local n = 18
hyper.tikzBegin("scale=5")
for k = 0, n - 1 do
  local P = complex.exp_i(2 * math.pi * k / n)
  local Q = complex.exp_i(2 * math.pi * (k + 1) / n)
  hyper.drawLine(P, Q, "ias boundary")
  hyper.tikzPrintf("\\node[stratum, label={[text=dzg singular, label distance=9mm]%f:$P_{%d}$}] at (%f, %f) {};",
    360 * k / n + 180, k, P.re, P.im)
end
hyper.tikzEnd()
\end{luacode*}
