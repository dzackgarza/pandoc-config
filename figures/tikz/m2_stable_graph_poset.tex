  \mathord{\vcenter{\hbox{%
    \begin{tikzpicture}[
      x=1em,
      y=1em,
      rootmtwo/.style={
        circle, draw, fill=white,
        inner sep=0pt, minimum size=5pt
      },
      rootmfour/.style={
        circle, draw, fill=black,
        inner sep=0pt, minimum size=5pt
      },
      mfour/.style={
        double,
        double distance=1.3pt,
        line width=0.35pt
      }
    ]
      \def\edgelen{1.15}

      % coordinates
      \coordinate (a0) at (0,0);
      \coordinate (a1) at (90:\edgelen);
      \coordinate (a2) at (210:\edgelen);
      \coordinate (a3) at (330:\edgelen);

      % edges first
      \draw[mfour] (a0) -- (a1);
      \draw[mfour] (a0) -- (a2);
      \draw[mfour] (a0) -- (a3);

      % vertices on top
      \node[rootmtwo]  at ($ (a0) - (0.15, 0)$) {};
      \node[rootmfour] at ($ (a1) - (0.15, 0)$) {};
      \node[rootmfour] at ($ (a2) - (0.15, 0)$) {};
      \node[rootmfour] at ($ (a3) - (0.15, 0)$) {};
    \end{tikzpicture}%
  }}}%
