% The diamond A B C D with centre M and edge midpoints P1..P4, cut into four
% hatched charts. Shared by fig_symington_16gon.tex and
% fig_geometric_degeneration.tex; \input inside a tikzpicture.
\coordinate (A) at (0, 2);   % top
\coordinate (B) at (-2, 0);  % left
\coordinate (C) at (0, -2);  % bottom
\coordinate (D) at (2, 0);   % right
\coordinate (M) at (0, 0);
\coordinate (P1) at ($(A)!0.5!(B)$);
\coordinate (P2) at ($(A)!0.5!(D)$);
\coordinate (P3) at ($(B)!0.5!(C)$);
\coordinate (P4) at ($(C)!0.5!(D)$);

\fill[ias chart=1] (A) -- (P1) -- (M) -- (P2) -- cycle;
\fill[ias chart=2] (B) -- (P1) -- (M) -- (P3) -- cycle;
\fill[ias chart=3] (C) -- (P3) -- (M) -- (P4) -- cycle;
\fill[ias chart=4] (D) -- (P2) -- (M) -- (P4) -- cycle;
\draw[ias boundary] (A) -- (B) -- (C) -- (D) -- cycle;
