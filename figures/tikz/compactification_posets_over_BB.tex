% Compactifications of F_T over the Baily-Borel compactification: the
% semitoroidal and toroidal posets of objects/posets/compactifications-over-BB
% (semitoroidal.json, toroidal.json), laid out by \posetfromjson. Inside each
% sub-poset the arrows point from the generic member to the more degenerate
% one; thick arrows are morphisms of compactifications.
\begin{tikzpicture}[font=\small]
  % Strictly non-toroidal semitoroidal compactifications.
  \posetfromjson[name prefix=S-, yshift=10cm, grade distance=1.65cm,
    element width=2.7cm, element height=0.9cm, element sep=1.3cm,
    element style={poset node, dashed}, cover style=degeneration,
    cover direction=up]{objects/posets/semitoroidal.json}
  % Toroidal compactifications; the Coxeter fan is highlighted.
  \posetfromjson[name prefix=T-, yshift=5cm, grade distance=1.5cm,
    element width=2.7cm, element height=0.9cm, element sep=1.8cm,
    element style=poset node, cover style=degeneration,
    cover direction=up]{objects/posets/toroidal.json}
  \node[poset node, minimum width=3.8cm, minimum height=0.9cm] (BB)
    at ([yshift=-4cm]T-2) {$\bbcpt{F_T}$};
  \node[poset node, minimum width=1.6cm, minimum height=0.9cm] (GIT)
    at ([xshift=-5cm, yshift=2cm]BB) {$\gitcpt{\mcm}$};
  \node[poset node, minimum width=1.6cm, minimum height=0.9cm] (KSBA)
    at ([xshift=5cm, yshift=2cm]BB) {$\ksbacpt{\mcm}$};

  \foreach \a/\b in {S-1/T-1, S-2/T-12, S-3/T-23, S-4/T-3, S-23/T-top,
      T-1/BB, T-2/BB, T-3/BB}
    \draw[morphism] (\a) -- (\b);
  \draw[morphism] (S-12) to[bend right=15] (T-12);
  \draw[morphism] (S-34) to[bend left=15] (T-23);
  \draw[morphism] (GIT) to[bend right=15] (BB);
  \draw[morphism] (KSBA) to[bend left=15] (BB);

  \begin{scope}[on background layer]
    \node[poset group, fill=dzg blob, inner sep=24pt,
          fit=(S-top)(S-1)(S-4)(T-1)(T-3)] (SemiBox) {};
    \node[poset group, fit=(T-1)(T-2)(T-3)(T-12)(T-23)(T-top)] (TorBox) {};
    \node[poset group, fill=dzg highlight!40,
          fit=(S-1)(S-2)(S-3)(S-4)(S-12)(S-23)(S-34)(S-top)] (NonTorBox) {};
  \end{scope}
  \begin{scope}[nodes={font=\scriptsize, inner sep=4pt}]
    \node[anchor=north west] at (SemiBox.north west) {Semitoroidal Compactifications};
    \node[anchor=north west] at (NonTorBox.north west) {Strictly Non-Toroidal};
    \node[anchor=south west] at (TorBox.south west) {Toroidal};
  \end{scope}
\end{tikzpicture}
