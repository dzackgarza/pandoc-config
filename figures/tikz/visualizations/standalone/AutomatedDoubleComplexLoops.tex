% The Serre spectral sequence of a fibration S^1 -> E -> CP^2:
% E_2^{p,q} = H^p(CP^2) (x) H^q(S^1), in four stages.
% Writing all E_2 terms
\begin{tikzpicture}
  \pic (E) [ss p name={$\CP^2$}, ss q name={$S^1$}, ss column sep=3.6cm] {ss page};
  \foreach \p in {0,...,5} \foreach \q in {0,...,5}
    {\ssentry{E}{\p}{\q}{$\ee{\p}{\q}$}}
\end{tikzpicture}

\vspace{1cm}

% Plugging in known groups: H^q(S^1) = Z for q = 0, 1 and 0 otherwise
\begin{tikzpicture}
  \pic (E) [ss p name={$\CP^2$}, ss q name={$S^1$}, ss column sep=3.2cm] {ss page};
  \foreach \q/\g in {0/\Z, 1/\Z, 2/0, 3/0, 4/0, 5/0} {
    \ssentry{E}{0}{\q}{$\zt{\Z}{\g}$}
    \foreach \p in {1,...,5} {\ssentry{E}{\p}{\q}{$\HCPtensor{\p}{\g}$}}}
\end{tikzpicture}

% Simplifying
\begin{tikzpicture}
  \pic (E) [ss p name={$\CP^2$}, ss q name={$S^1$}, ss column sep=1.6cm] {ss page};
  \foreach \q in {0,1} {
    \ssentry{E}{0}{\q}{$\Z$}
    \foreach \p in {1,...,5} {\ssentry{E}{\p}{\q}{$\HCP{\p}$}}}
  \foreach \q in {2,...,5} \foreach \p in {0,...,5} {\ssentry{E}{\p}{\q}{$0$}}
\end{tikzpicture}

% The differentials d_2: E_2^{p,1} -> E_2^{p+2,0}
\begin{tikzpicture}
  \pic (E) [ss p name={$\CP^2$}, ss q name={$S^1$}, ss p={0,...,6},
    ss column sep=1.6cm] {ss page};
  \foreach \q in {0,1} {
    \ssentry{E}{0}{\q}{$\Z$}
    \foreach \p in {1,...,5} {\ssentry{E}{\p}{\q}{$\HCP{\p}$}}
    \ssentry{E}{6}{\q}{$\cdots$}}
  \foreach \q in {2,...,5} \foreach \p in {0,...,5} {\ssentry{E}{\p}{\q}{$0$}}
  \foreach \p in {0,...,4} {\ssdifferential[ss unlabelled]{E}{2}{\p}{1}}
\end{tikzpicture}
