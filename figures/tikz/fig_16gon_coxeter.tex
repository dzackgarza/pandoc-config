% Coxeter diagram attached to a 16-gon: sixteen boundary vertices b_1..b_16
% around a square, four interior vertices l_16..l_19 joined to the corners,
% and l_20, l_21 joined by a weight-infinity edge.
\begin{tikzpicture}
  \foreach \i in {1, ..., 5} \node[vertex=white] (b\i) at (\i - 1, 0) {};
  \foreach \i in {6, 7, 8} \node[vertex=white] (b\i) at (4, \i - 5) {};
  \foreach \i in {9, ..., 13} \node[vertex=white] (b\i) at (13 - \i, 4) {};
  \foreach \i in {14, 15, 16} \node[vertex=white] (b\i) at (0, 17 - \i) {};
  \foreach \v/\x/\y/\l in {19/1/3/below left, 18/3/3/below right, 16/1/1/above left,
                          17/3/1/above right}
    \node[vertex=white, label={\l:$l_{\v}$}] (l\v) at (\x, \y) {};
  \node[vertex=black, label={right:$l_{20}$}] (l20) at (2, 2.5) {};
  \node[vertex=black, label={right:$l_{21}$}] (l21) at (2, 1.5) {};
  \begin{scope}[on edge layer]
    \draw[coxeter edge=3] (l19.center) -- (l20.center) -- (l18.center)
      (l16.center) -- (l21.center) -- (l17.center)
      (b13.center) -- (l19.center) (b9.center) -- (l18.center)
      (b1.center) -- (l16.center) (b5.center) -- (l17.center);
    \draw[coxeter edge=infinity] (l20.center) -- (l21.center);
  \end{scope}
\end{tikzpicture}
