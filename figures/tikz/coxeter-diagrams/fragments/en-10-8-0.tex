% Coxeter diagram of the Enriques lattice (10,8,0)_1 in Bourbaki numbering:
% the chain 1,3,4,...,9 with 2 attached to 4, and the root 10 of the other
% square joined to 9 by a weight-4 edge.
% Picture body: \input it inside a tikzpicture or a shifted scope.
\foreach \v/\x in {1/0, 3/1, 4/2, 5/3, 6/4, 7/5, 8/6, 9/7}
  \node[vertex=black, label={[vertex label]below:$\v$}] (\v) at (\x, 0) {};
\node[vertex=black, label={[vertex label]above:$2$}] (2) at (2, 1) {};
\node[vertex=white, label={[vertex label]below:$10$}] (10) at (8, 0) {};
\begin{scope}[on edge layer]
  \draw[coxeter edge=3] (1.center) -- (9.center) (2.center) -- (4.center);
  \draw[coxeter edge=4] (9.center) -- (10.center);
\end{scope}
