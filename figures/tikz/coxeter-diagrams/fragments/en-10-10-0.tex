% Coxeter diagram of the Enriques lattice (10,10,0)_1, i.e. E_10, in Bourbaki
% numbering: the chain 1,3,4,...,10 with 2 attached to 4.
% Picture body: \input it inside a tikzpicture or a shifted scope.
\foreach \v/\x in {1/0, 3/1, 4/2, 5/3, 6/4, 7/5, 8/6, 9/7, 10/8}
  \node[vertex=black, label={[vertex label]below:$\v$}] (\v) at (\x, 0) {};
\node[vertex=black, label={[vertex label]above:$2$}] (2) at (2, 1) {};
\begin{scope}[on edge layer]
  \draw[coxeter edge=3] (1.center) -- (10.center) (2.center) -- (4.center);
\end{scope}
