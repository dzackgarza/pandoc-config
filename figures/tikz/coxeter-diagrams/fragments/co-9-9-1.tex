% Coxeter diagram of the Coble lattice (9,9,1)_1 in Bourbaki numbering: the
% chain 1,3,4,...,8 with 2 attached to 4, and the root 9 of the other square
% joined to 8 by a weight-4 edge.
% Picture body: \input it inside a tikzpicture or a shifted scope.
\foreach \v/\x in {1/0, 3/1, 4/2, 5/3, 6/4, 7/5, 8/6}
  \node[vertex=black, label={[vertex label]below:$\v$}] (\v) at (\x, 0) {};
\node[vertex=black, label={[vertex label]above:$2$}] (2) at (2, 1) {};
\node[vertex=white, label={[vertex label]below:$9$}] (9) at (7, 0) {};
\begin{scope}[on edge layer]
  \draw[coxeter edge=3] (1.center) -- (8.center) (2.center) -- (4.center);
  \draw[coxeter edge=4] (8.center) -- (9.center);
\end{scope}
